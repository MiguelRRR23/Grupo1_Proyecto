CREATE PROCEDURE sp_ListarCuentas()
SELECT 
    c.numero_cuenta,
    c.dni,
    p.NOMBRE AS nombre,
    p.APELLIDO AS apellido,
    c.tipo_cuenta,
    c.monto,
    c.banco
FROM cuentas c
INNER JOIN personas p ON c.dni = p.DNI;


CREATE PROCEDURE sp_BuscarCuentaDNI(IN dni_buscar INT)
SELECT 
    c.numero_cuenta,
    c.dni,
    p.NOMBRE AS nombre,
    p.APELLIDO AS apellido,
    c.tipo_cuenta,
    c.monto,
    c.banco
FROM cuentas c
INNER JOIN personas p ON c.dni = p.DNI
WHERE c.dni = dni_buscar;


CREATE PROCEDURE sp_InsertarCuenta(
    IN num VARCHAR(20),
    IN dni_cliente INT,
    IN tipo VARCHAR(30),
    IN monto DECIMAL(10,2),
    IN banco VARCHAR(100)
)
INSERT INTO cuentas VALUES (num, dni_cliente, tipo, monto, banco);


CREATE PROCEDURE sp_ModificarCuenta(
    IN num VARCHAR(20),
    IN tipo VARCHAR(30),
    IN monto DECIMAL(10,2)
)
UPDATE cuentas 
SET tipo_cuenta = tipo,
    monto = monto
WHERE numero_cuenta = num;


CREATE PROCEDURE sp_EliminarCuenta(IN cuenta VARCHAR(20))
DELETE FROM cuentas WHERE numero_cuenta = cuenta;
