--
-- Tabla `horarios`
--
-- Esta tabla almacena los turnos disponibles para los empleados.
-- Se compone de un identificador, el día o días que abarca el turno
-- y el rango horario en formato de texto (por ejemplo, '08:00-14:00').

CREATE TABLE horarios (
    id_turno INT AUTO_INCREMENT PRIMARY KEY,
    dia_turnos VARCHAR(50),
    horario VARCHAR(50)
);

-- turnos típicos; pueden ampliarse o modificarse según se requiera.

('Lunes a Viernes', '10:00-18:00'),
('Sabado y Domingo', '09:00-13:00');

CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    usuario VARCHAR(50) UNIQUE,
    clave VARCHAR(100),
    rol VARCHAR(20),
    -- id_turno referencia el turno asignado al usuario. Puede ser NULL para usuarios
    -- con rol ADMIN, ya que no están sujetos a un horario específico.
    id_turno INT,
    FOREIGN KEY (id_turno) REFERENCES horarios(id_turno)
);

CREATE PROCEDURE sp_ListarCuentas()
SELECT 
    c.numero_cuenta,
    c.dni_persona AS dni,
    p.nombre AS nombre,
    p.apellido AS apellido,
    c.tipo_cuenta,
    c.monto,
    c.num_cuotas,
    c.fecha_inicio,
    c.fecha_fin
FROM cuentas c
INNER JOIN personas p ON c.dni_persona = p.dni_persona;


CREATE PROCEDURE sp_BuscarCuentaDNI(IN dni_buscar INT)
SELECT 
    c.numero_cuenta,
    c.dni_persona AS dni,
    p.nombre AS nombre,
    p.apellido AS apellido,
    c.tipo_cuenta,
    c.monto,
    c.num_cuotas,
    c.fecha_inicio,
    c.fecha_fin
FROM cuentas c
INNER JOIN personas p ON c.dni_persona = p.dni_persona
WHERE c.dni_persona = dni_buscar;


CREATE PROCEDURE sp_InsertarCuenta(
    IN num VARCHAR(20),
    IN dni_in INT,
    IN tipo VARCHAR(30),
    IN monto DECIMAL(10,2),
    IN num_cuotas INT,
    IN fecha_inicio DATE,
    IN fecha_fin DATE
)
INSERT INTO cuentas (numero_cuenta, dni_persona, tipo_cuenta, monto, num_cuotas, fecha_inicio, fecha_fin)
VALUES (num, dni_in, tipo, monto, num_cuotas, fecha_inicio, fecha_fin);


CREATE PROCEDURE sp_ModificarCuenta(
    IN num VARCHAR(20),
    IN tipo VARCHAR(30),
    IN monto DECIMAL(10,2),
    IN num_cuotas INT,
    IN fecha_inicio DATE,
    IN fecha_fin DATE
)
UPDATE cuentas
SET tipo_cuenta = tipo,
    monto = monto,
    num_cuotas = num_cuotas,
    fecha_inicio = fecha_inicio,
    fecha_fin = fecha_fin
WHERE numero_cuenta = num;


CREATE PROCEDURE sp_EliminarCuenta(IN cuenta VARCHAR(20))
DELETE FROM cuentas WHERE numero_cuenta = cuenta;

--
-- Tabla `prestamos`
--
-- Representa cada préstamo otorgado. Establece una relación de muchos-a-muchos
-- entre clientes y usuarios (prestamistas), y referencia la cuenta en la que se
-- depositará el préstamo. Contiene datos de evaluación (monto solicitado,
-- número de cuotas, tasa anual, ingresos mensuales, motivo) así como las
-- fechas de inicio y fin del pago.

CREATE TABLE prestamos (
    id_prestamo INT AUTO_INCREMENT PRIMARY KEY,
    dni_persona INT,
    id_usuario INT,
    numero_cuenta VARCHAR(20),
    monto DECIMAL(10,2),
    num_cuotas INT,
    tasa_anual DECIMAL(5,2),
    fecha_inicio DATE,
    fecha_fin DATE,
    estado VARCHAR(20),
    motivo VARCHAR(255),
    ingresos_mensuales DECIMAL(10,2),
    FOREIGN KEY (dni_persona) REFERENCES personas(dni_persona),
    -- La columna id_usuario referencia a la tabla usuarios, ya que
    -- los prestamistas son también usuarios del sistema. De esta
    -- manera se mantiene la relación muchos-a-muchos entre clientes y
    -- empleados.
    FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario),
    FOREIGN KEY (numero_cuenta) REFERENCES cuentas(numero_cuenta) ON DELETE CASCADE
);

-- Procedimientos para la gestión de préstamos
CREATE PROCEDURE sp_ListarPrestamos()
SELECT 
    pr.id_prestamo,
    pr.dni_persona AS dni,
    p.nombre AS nombre_cliente,
    p.apellido AS apellido_cliente,
    pr.id_usuario,
    u.usuario AS nombre_usuario,
    pr.numero_cuenta,
    pr.monto,
    pr.num_cuotas,
    pr.tasa_anual,
    pr.fecha_inicio,
    pr.fecha_fin,
    pr.estado,
    pr.motivo,
    pr.ingresos_mensuales
FROM prestamos pr
INNER JOIN personas p ON pr.dni_persona = p.dni_persona
LEFT JOIN usuarios u ON pr.id_usuario = u.id_usuario;

CREATE PROCEDURE sp_ListarPrestamosPorCliente(IN dni_buscar INT)
SELECT 
    id_prestamo,
    dni_persona AS dni,
    id_usuario,
    numero_cuenta,
    monto,
    num_cuotas,
    tasa_anual,
    fecha_inicio,
    fecha_fin,
    estado,
    motivo,
    ingresos_mensuales
FROM prestamos 
WHERE dni_persona = dni_buscar;

-- Lista los préstamos registrados por un usuario específico
CREATE PROCEDURE sp_ListarPrestamosPorUsuario(IN id_buscar INT)
SELECT 
    pr.id_prestamo,
    pr.dni_persona AS dni,
    pr.id_usuario,
    pr.numero_cuenta,
    pr.monto,
    pr.num_cuotas,
    pr.tasa_anual,
    pr.fecha_inicio,
    pr.fecha_fin,
    pr.estado,
    pr.motivo,
    pr.ingresos_mensuales
FROM prestamos pr
WHERE pr.id_usuario = id_buscar;

CREATE PROCEDURE sp_InsertarPrestamo(
    IN dni_in INT,
    IN id_usuario_in INT,
    IN numero_cuenta VARCHAR(20),
    IN monto DECIMAL(10,2),
    IN num_cuotas INT,
    IN tasa_anual DECIMAL(5,2),
    IN fecha_inicio DATE,
    IN fecha_fin DATE,
    IN estado VARCHAR(20),
    IN motivo VARCHAR(255),
    IN ingresos_mensuales DECIMAL(10,2)
)
INSERT INTO prestamos (
    dni_persona,
    id_usuario,
    numero_cuenta,
    monto,
    num_cuotas,
    tasa_anual,
    fecha_inicio,
    fecha_fin,
    estado,
    motivo,
    ingresos_mensuales
)
VALUES (
    dni_in,
    id_usuario_in,
    numero_cuenta,
    monto,
    num_cuotas,
    tasa_anual,
    fecha_inicio,
    fecha_fin,
    estado,
    motivo,
    ingresos_mensuales
);

CREATE PROCEDURE sp_ModificarPrestamo(
    IN id_prestamo_in INT,
    IN monto DECIMAL(10,2),
    IN num_cuotas INT,
    IN tasa_anual DECIMAL(5,2),
    IN fecha_inicio DATE,
    IN fecha_fin DATE,
    IN estado VARCHAR(20),
    IN motivo VARCHAR(255),
    IN ingresos_mensuales DECIMAL(10,2)
)
UPDATE prestamos
SET monto = monto,
    num_cuotas = num_cuotas,
    tasa_anual = tasa_anual,
    fecha_inicio = fecha_inicio,
    fecha_fin = fecha_fin,
    estado = estado,
    motivo = motivo,
    ingresos_mensuales = ingresos_mensuales
WHERE id_prestamo = id_prestamo_in;

CREATE PROCEDURE sp_EliminarPrestamo(IN id_prestamo_in INT)
DELETE FROM prestamos WHERE id_prestamo = id_prestamo_in;
